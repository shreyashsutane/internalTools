A. Dynamic {{variable}} Extractor & Resolver superadmin/src/sql-formatter.ts 
// Extracts all unique {{variable_name}} tokens
export const extractVariables = (sql: string): string[] => {
    if (!sql || typeof sql !== 'string') return [];
    const regex = /\{\{\s*([a-zA-Z0-9_]+)\s*\}\}/g;
    const matches = new Set<string>();
    let match: RegExpExecArray | null;
    while ((match = regex.exec(sql)) !== null) {
        if (match[1]) matches.add(match[1].trim());
    }
    return Array.from(matches);
};

// Replaces {{variable_name}} with user input values
export const resolveSqlVariables = (sql: string, values: Record<string, string>): string => {
    if (!sql || typeof sql !== 'string') return '';
    return sql.replace(/\{\{\s*([a-zA-Z0-9_]+)\s*\}\}/g, (_full, varName) => {
        const val = values[varName.trim()];
        return val !== undefined ? val : '';
    });
};
B. Automated updatedByName & ISO Timestamp Tracking superadmin/src/api.ts 
// Converts shreyash.sutane@gmail.com -> "Shreyash Sutane"
export const formatEmailToDisplayName = (email: string): string => {
    const localPart = (email || '').split('@')[0] || '';
    const words = localPart.replace(/[._\-+]/g, ' ').trim().split(/\s+/);
    return words.map(w => w.charAt(0).toUpperCase() + w.slice(1).toLowerCase()).join(' ');
};

// Atomically commits updated SQL query and user audit metadata to Datastore
export const saveQuestionEntity = async (projectId, question, newSql, extraProps) => {
    const properties = { ...question.properties };
    properties[question.queryField] = { stringValue: newSql };
    properties.updatedByName = { stringValue: State.userName }; // e.g. "Shreyash Sutane"
    properties.updatedAt = { timestampValue: new Date().toISOString() };
    
    // Commit upsert mutation directly to Datastore REST
    await fetch(`https://datastore.googleapis.com/v1/projects/${projectId}:commit`, {
        method: 'POST',
        headers: { Authorization: `Bearer ${State.token}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({ mode: 'NON_TRANSACTIONAL', mutations: [{ upsert: { key: question.key, properties } }] })
    });
};
C. Memory-Safe 100,000-Row Chunked CSV Export superadmin/src/utils.ts 
export const exportCsv = (filename: string, headers: string[], rows: any[][]) => {
    const parts: string[] = [];
    parts.push(headers.map(h => `"${String(h).replace(/"/g, '""')}"`).join(',') + '\n');

    const total = rows.length;
    const chunkSize = 10000;

    // Stream through multi-part array buffer to prevent V8 string memory limits
    for (let i = 0; i < total; i += chunkSize) {
        const chunk = rows.slice(i, i + chunkSize);
        const chunkLines = chunk.map(row =>
            row.map(cell => cell === null ? '""' : `"${String(cell).replace(/"/g, '""')}"`).join(',')
        );
        parts.push(chunkLines.join('\n') + '\n');
    }

    const blob = new Blob(parts, { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `${filename}.csv`;
    link.click();
    URL.revokeObjectURL(url);
};
